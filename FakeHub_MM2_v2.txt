local function charParts(p)
	local c = (p or plr).Character
	if not c then return end
	return c, c:FindFirstChildOfClass("Humanoid"), c:FindFirstChild("HumanoidRootPart")
end

local function hasTool(p, name)
	local c = p.Character
	local b = p:FindFirstChild("Backpack")
	return (c and c:FindFirstChild(name)) or (b and b:FindFirstChild(name))
end

local function roleOf(p)
	if hasTool(p, KNIFE_NAME) then return "Murderer" end
	if hasTool(p, GUN_NAME) then return "Sheriff" end
	return "Innocent"
end

local function weaponOf(p)
	local c = p.Character
	if c then for _, t in ipairs(c:GetChildren()) do if t:IsA("Tool") then return t.Name end end end
	local b = p:FindFirstChild("Backpack")
	if b then for _, t in ipairs(b:GetChildren()) do if t:IsA("Tool") then return t.Name end end end
	return "-"
end

local function equipTool(name)
	local c, hum = charParts()
	if not (c and hum) then return end
	local t = c:FindFirstChild(name)
	if t then return t end
	local bp = plr:FindFirstChild("Backpack")
	local inBp = bp and bp:FindFirstChild(name)
	if inBp then hum:EquipTool(inBp) return inBp end
end

local function newLine()
	return new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0, Visible = false}, espGui)

local function setLine(f, a, b, thick, col, alpha)
	local d = b - a
	f.Size = UDim2.fromOffset(d.Magnitude, thick)
	f.Position = UDim2.fromOffset((a.X + b.X) / 2, (a.Y + b.Y) / 2)
	f.Rotation = math.deg(math.atan2(d.Y, d.X))
	f.BackgroundColor3 = col
	f.BackgroundTransparency = alpha
	f.Visible = true
end

local function updatePlayerESP(p, o, myRoot, rainbow)
	local c, h, r = charParts(p)
	local hl = c and c:FindFirstChild("FHHighlight")
	if not (C.esp and c and h and r and h.Health > 0) then
		hideObjs(o)
		if hl then hl:Destroy() end
		return
	end
	local dist = (r.Position - myRoot.Position).Magnitude
	if dist > C.espDist then
		hideObjs(o)
		if hl then hl:Destroy() end
		return
	end
	local role = roleOf(p)
	local col = C.rainbow and rainbow or (COLORS[C[ROLEKEY[role]]] or WHITE)
	if C.visCheck and not isVisible(r, c) then col = col:Lerp(Color3.new(0, 0, 0), 0.5) end
	local alpha = C.espAlpha / 100

	if C.chams then
		if not hl then
			hl = new("Highlight", {Name = "FHHighlight", DepthMode = Enum.HighlightDepthMode.AlwaysOnTop, FillTransparency = 0.5}, c)
		end
		hl.FillColor = col
		hl.OutlineColor = col
	elseif hl then
		hl:Destroy()
	end

	local rp, on = cam:WorldToViewportPoint(r.Position)
	if not on then hideObjs(o) return end
	local top = cam:WorldToViewportPoint(r.Position + Vector3.new(0, 3, 0))
	local bot = cam:WorldToViewportPoint(r.Position - Vector3.new(0, 3.2, 0))
	local hgt = math.abs(bot.Y - top.Y)
	local wid = hgt * 0.55
	local thick = C.lineThick

	if C.box then
		o.box.Visible = true
		o.box.Position = UDim2.fromOffset(rp.X, (top.Y + bot.Y) / 2)
		o.box.Size = UDim2.fromOffset(wid, hgt)
		o.boxStroke.Color = col
		o.boxStroke.Thickness = thick
		o.boxStroke.Transparency = alpha
	else
		o.box.Visible = false
	end

	if C.tracers then
		setLine(o.tracer, Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y), Vector2.new(rp.X, bot.Y), thick, col, alpha)
	else
		o.tracer.Visible = false
	end

	local head = c:FindFirstChild("Head")
	if C.headCircle and head then
		local hp = cam:WorldToViewportPoint(head.Position)
		local rad = math.max(4, hgt * 0.09)
		o.head.Visible = true
		o.head.Position = UDim2.fromOffset(hp.X, hp.Y)
		o.head.Size = UDim2.fromOffset(rad * 2, rad * 2)
		o.headStroke.Color = col
		o.headStroke.Thickness = thick
		o.headStroke.Transparency = alpha
	else
		o.head.Visible = false
	end

	local used = 0
	if C.skeleton and not C.lowPerf then
		local list = (h.RigType == Enum.HumanoidRigType.R15) and R15 or R6
		for i, pair in ipairs(list) do
			local pa, pb = c:FindFirstChild(pair[1]), c:FindFirstChild(pair[2])
			if pa and pb then
				local a, aOn = cam:WorldToViewportPoint(pa.Position)
				local b, bOn = cam:WorldToViewportPoint(pb.Position)
				if aOn and bOn then
					setLine(o.skel[i], Vector2.new(a.X, a.Y), Vector2.new(b.X, b.Y), thick, col, alpha)
					used = i
				else
					o.skel[i].Visible = false
				end
			else
				o.skel[i].Visible = false
			end
		end
	end
	for i = used + 1, 14 do o.skel[i].Visible = false end

	local parts = {}
	if C.names then table.insert(parts, p.Name) end
	if C.roles then table.insert(parts, "[" .. role .. "]") end
	if C.weapons then table.insert(parts, weaponOf(p)) end
	if C.health then table.insert(parts, math.floor(h.Health) .. "hp") end
	if C.dist then table.insert(parts, math.floor(dist) .. "m") end
	if #parts > 0 then
		o.text.Visible = true
		o.text.Text = table.concat(parts, " | ")
		o.text.Position = UDim2.fromOffset(rp.X, top.Y - 2)
		o.text.TextColor3 = col
		o.text.TextSize = C.textSize
		o.text.Font = Enum.Font[C.espFont] or Enum.Font.GothamBold
		o.text.TextStrokeTransparency = C.textStroke and 0.2 or 1
		o.text.TextTransparency = alpha
	else
		o.text.Visible = false
	end
end

local function updateRadar(myRoot, rainbow)
	radar.Visible = C.radar
	if not C.radar then return end
	local size = C.radarSize
	radar.Size = UDim2.fromOffset(size, size)
	radarCorner.CornerRadius = UDim.new(C.radarShape == "Circle" and 1 or 0, 0)
	local R = size / 2
	radarSelf.Position = UDim2.fromOffset(R, R)
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= plr then
			local o = espObjs[p] or mkObjs(p)
			local c, h, r = charParts(p)
			if c and h and r and h.Health > 0 and myRoot then
				local rel = cam.CFrame:VectorToObjectSpace(r.Position - myRoot.Position)
				local v = Vector2.new(rel.X, rel.Z) / C.radarRange * R
				if C.radarShape == "Circle" then
					if v.Magnitude > R - 4 then v = v.Unit * (R - 4) end
				else
					v = Vector2.new(math.clamp(v.X, -(R - 4), R - 4), math.clamp(v.Y, -(R - 4), R - 4))
				end
				o.dot.Position = UDim2.fromOffset(R + v.X, R + v.Y)
				o.dot.BackgroundColor3 = C.rainbow and rainbow or (COLORS[C[ROLEKEY[roleOf(p)]]] or WHITE)
				o.dot.Visible = true
			else
				o.dot.Visible = false
			end
		end
	end
end

local T = {shoot = 0, stab = 0, throw = 0}

local function getTarget(ignoreFov)
	local bestPos, bestP, bestPart, bd = nil, nil, nil, ignoreFov and math.huge or C.fov
	local _, _, myRoot = charParts()
	if not myRoot then return end
	local center = cam.ViewportSize / 2
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= plr then
			local c, h, r = charParts(p)
			if c and h and r and h.Health > 0 and (C.aimTarget == "Everyone" or roleOf(p) == "Murderer") then
				local part = c:FindFirstChild(C.aimPart) or c:FindFirstChild("Head") or r
				if part and (part.Position - myRoot.Position).Magnitude <= C.maxDist then
					local pos = part.Position
					if C.predict then pos = pos + r.AssemblyLinearVelocity * 0.1 end
					local v, on = cam:WorldToViewportPoint(pos)
					if on then
						local d = (Vector2.new(v.X, v.Y) - center).Magnitude
						if d < bd and (not C.visAim or isVisible(part, c)) then
							bestPos, bestP, bestPart, bd = pos, p, part, d
						end
					end
				end
			end
		end
	end
	return bestPos, bestP, bestPart, bd
end

local function nearest(range, needVisible)
	local _, _, myRoot = charParts()
	if not myRoot then return end
	local best, bd, bp = nil, range, nil
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= plr then
			local c, h, r = charParts(p)
			if c and h and r and h.Health > 0 then
				local d = (r.Position - myRoot.Position).Magnitude
				if d < bd and (not needVisible or isVisible(r, c)) then best, bd, bp = r, d, p end
			end
		end
	end
	return best, bp
end

function A.shoot()
	local gun = equipTool(GUN_NAME)
	if gun then gun:Activate() end
end
function A.fire()
	local pos = getTarget(true)
	if pos then cam.CFrame = CFrame.lookAt(cam.CFrame.Position, pos) end
	A.shoot()
end
function A.stab()
	local _, _, root = charParts()
	if not root then return end
	local knife = equipTool(KNIFE_NAME)
	if not knife then return end
	local r = nearest(C.stabTp and C.tpRange or C.stabRange)
	if r then
		if C.stabTp and (r.Position - root.Position).Magnitude > C.stabRange then
			root.CFrame = r.CFrame * CFrame.new(0, 0, 2.5)
		end
		if C.faceTarget then
			root.CFrame = CFrame.lookAt(root.Position, Vector3.new(r.Position.X, root.Position.Y, r.Position.Z))
		end
	end
	knife:Activate()
end
function A.throw()
	local r = nearest(C.throwRange, true)
	if not r then return end
	local knife = equipTool(KNIFE_NAME)
	if not knife then return end
	cam.CFrame = CFrame.lookAt(cam.CFrame.Position, r.Position)
	knife:Activate()
end

			end
		end
	end
	return names
end

local function collect(name, flagKey)
	local _, _, root = charParts()
	if not root then return 0 end
	local back = root.CFrame
	local n = 0
	for _, d in ipairs(workspace:GetDescendants()) do
		if d:IsA("BasePart") and d.Name == name and d.Parent then
			if not root.Parent or not C[flagKey] then break end
			root.CFrame = d.CFrame
			task.wait(0.12)
			n += 1
			if n >= C.farmMax then break end
		end
	end
	if root.Parent then root.CFrame = back end
	return n
end

local function slider(page, name, min, max, key, cb)
	local f = item(page, 44, name)
	local lbl = label(f, "", 4)
	local track = new("Frame", {Size = UDim2.new(1, -16, 0, 10), Position = UDim2.new(0, 8, 0, 27), BackgroundColor3 = LIGHT}, f)
	corner(track, 5)
	local fill = new("Frame", {Size = UDim2.new(0, 0, 1, 0)}, track)
	corner(fill, 5)
	themed(function() fill.BackgroundColor3 = accent() end)
	local function refresh()
		lbl.Text = name .. ": " .. tostring(C[key])
		fill.Size = UDim2.new(math.clamp((C[key] - min) / (max - min), 0, 1), 0, 1, 0)
	end
	reg[key] = refresh
	refresh()
	local active = false
	local function update(x)
		local r = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		C[key] = math.floor(min + (max - min) * r + 0.5)
		refresh()
		if cb then cb(C[key]) end
	end
	track.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			active = true
			update(i.Position.X)
		end
	end)
	UIS.InputChanged:Connect(function(i)
		if active and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			update(i.Position.X)
		end
	end)
	UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			active = false
		end
	end)
end

local function buildSearch(q)
	local sp = pages["Search"].page
	for _, c in ipairs(sp:GetChildren()) do if c:IsA("GuiObject") then c:Destroy() end end
	q = q:lower()
	local n = 0
	for _, it in ipairs(allItems) do
		if it.name:lower():find(q, 1, true) and n < 40 then
			n += 1
			local b = new("TextButton", {Size = UDim2.new(1, -8, 0, 30), BackgroundColor3 = MID, TextColor3 = WHITE,
				Text = it.name .. "   [" .. it.tab .. "]", Font = Enum.Font.Gotham, TextSize = 12, LayoutOrder = n}, sp)
			corner(b, 8)
			b.Activated:Connect(function()
				lastPage = it.tab
				searchBox.Text = ""
				showPage(it.tab)
			end)
		end
	end
	if n == 0 then
		new("TextLabel", {Size = UDim2.new(1, -8, 0, 30), BackgroundTransparency = 1, Text = "Нічого не знайдено",
			TextColor3 = WHITE, Font = Enum.Font.Gotham, TextSize = 12}, sp)
	end
end
searchBox:GetPropertyChangedSignal("Text"):Connect(function()
	if searchBox.Text == "" then showPage(lastPage) else buildSearch(searchBox.Text) showPage("Search") end
end)

local fpsSmooth, frameN, lastFov, lastGrav = 60, 0, nil, 196
local bv, wasGhost = nil, false

RS.RenderStepped:Connect(function(dt)
	cam = workspace.CurrentCamera
	fpsSmooth = fpsSmooth * 0.9 + (1 / math.max(dt, 0.001)) * 0.1
	if C.fovCam ~= lastFov then cam.FieldOfView = C.fovCam lastFov = C.fovCam end

	circle.Visible = C.aim and C.fovCircle
	if circle.Visible then
		circle.Size = UDim2.fromOffset(C.fov * 2, C.fov * 2)
		circleStroke.Color = COLORS[C.fovColor] or COLORS.Blue
	end

	local previewPart
	if C.aim or C.autoShoot or C.hitPreview then
		local pos, _, part, sd = getTarget(false)
		if pos then
			previewPart = part
			if C.aim then
				cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, pos), math.clamp(C.smooth / 100, 0.03, 1))
			end
			if C.autoShoot and tick() - T.shoot > C.shootDelay / 10 and ((not C.aim) or sd < 70) then
				T.shoot = tick()
				A.shoot()
			end
		end
	end
	previewAdorn.Adornee = previewPart
	previewAdorn.Visible = C.hitPreview and previewPart ~= nil

	frameN += 1
	local every = C.lowPerf and math.max(3, C.espEvery) or C.espEvery
	local _, _, myRoot = charParts()
	local rainbow = Color3.fromHSV(tick() % 5 / 5, 1, 1)
	if myRoot and frameN % every == 0 then
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= plr then
				local o = espObjs[p] or mkObjs(p)
				updatePlayerESP(p, o, myRoot, rainbow)
			end
		end
	elseif not myRoot then
		for _, o in pairs(espObjs) do hideObjs(o) end
	end
	updateRadar(myRoot, rainbow)
end)

RS.Heartbeat:Connect(function()
	if C.gravity ~= lastGrav then workspace.Gravity = C.gravity lastGrav = C.gravity end
	local char, hum, root = charParts()
	if not (char and hum and root) then return end

	hum.WalkSpeed = C.speed
	hum.UseJumpPower = true
	hum.JumpPower = C.jump * (C.highJump and 2 or 1)

	-- fly
	if C.fly then
		if not bv or bv.Parent ~= root then
			if bv then bv:Destroy() end
			bv = new("BodyVelocity", {MaxForce = Vector3.new(1e5, 1e5, 1e5), Velocity = Vector3.zero}, root)
		end
		local dir = hum.MoveDirection
		local vert = (UIS:IsKeyDown(Enum.KeyCode.Space) and 1 or 0) - (UIS:IsKeyDown(Enum.KeyCode.LeftControl) and 1 or 0)
		if dir.Magnitude > 0 then
			local look = cam.CFrame.LookVector
			local flat = Vector3.new(look.X, 0, look.Z)
			local sign = (flat.Magnitude > 0 and dir:Dot(flat.Unit) < 0) and -1 or 1
			bv.Velocity = look * C.flySpeed * sign + Vector3.new(0, vert * C.flySpeed, 0)
		else
			bv.Velocity = Vector3.new(0, vert * C.flySpeed, 0)
		end
	elseif bv then
		bv:Destroy()
		bv = nil
	end

	-- spin
	if C.spin then
		local a = math.rad(C.spinSpeed)
		if C.spinMode == "Right" then a = -a
		elseif C.spinMode == "Jitter" then a = math.rad(math.random(-C.spinSpeed * 4, C.spinSpeed * 4)) end
		root.CFrame = root.CFrame * CFrame.Angles(0, a, 0)
	end

	-- bhop
	if C.bhop and UIS:IsKeyDown(Enum.KeyCode.Space) and hum.FloorMaterial ~= Enum.Material.Air then
		local v = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(v.X, C.bhopPower, v.Z)
	end

	-- напівпрозорий
	if C.ghost or wasGhost then
		for _, p in ipairs(char:GetDescendants()) do
			if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then p.Transparency = C.ghost and 0.6 or 0 end
		end
		wasGhost = C.ghost
	end

	-- автоудар / автокидок
	if C.stab and tick() - T.stab > C.stabDelay / 10 then
		if nearest(C.stabTp and C.tpRange or C.stabRange) then
			T.stab = tick()
			A.stab()
		end
	end
	if C.autoThrow and tick() - T.throw > C.throwDelay / 10 then
		if nearest(C.throwRange, true) and not nearest(C.stabRange) then
			T.throw = tick()
			A.throw()
		end
	end
end)

	local _, hum, root = charParts()
	if not (hum and root) then return end
	if C.infJump then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
	if C.wallHop then
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {plr.Character}
		params.FilterType = Enum.RaycastFilterType.Exclude
		if workspace:Raycast(root.Position, root.CFrame.LookVector * 3, params) then
			local v = root.AssemblyLinearVelocity
			root.AssemblyLinearVelocity = Vector3.new(v.X, C.bhopPower, v.Z)
		end
	end
end)
